## 设计目标

它不是 Transformer 的 KV Cache，而是**长期结构记忆（Long-term Structural Memory）**。

它要完成四件事：

1. 接收当前层的隐藏状态 `H`
2. 与历史记忆 `M` 融合
3. 压缩成固定大小的长期记忆
4. 输出新的记忆供下一层使用

数据流如下：

```
Hidden State
      │
      ▼
+----------------+
| Spiral Memory  |
+----------------+
      │
      ├────────► New Memory
      │
      ▼
Fused Hidden State
```

# 1. spiral_memory.py

```Python
import torch
import torch.nn as nn
import torch.nn.functional as F


class SpiralMemory(nn.Module):
    """
    Information Spiral Memory v0.1

    功能:
    Hidden State
        ↓
    Encode
        ↓
    Compress
        ↓
    Update Memory

    """

    def __init__(
        self,
        hidden_size,
        memory_slots=32
    ):
        super().__init__()

        self.hidden_size = hidden_size
        self.memory_slots = memory_slots


        # 信息编码器
        self.encoder = nn.Sequential(
            nn.Linear(hidden_size, hidden_size),
            nn.GELU(),
            nn.Linear(hidden_size, hidden_size)
        )


        # Memory Attention
        self.memory_attention = nn.MultiheadAttention(
            embed_dim=hidden_size,
            num_heads=8,
            batch_first=True
        )


        # 压缩器
        self.compressor = nn.Linear(
            hidden_size,
            memory_slots
        )


        # 更新门
        self.update_gate = nn.Sequential(
            nn.Linear(hidden_size * 2, hidden_size),
            nn.Sigmoid()
        )


    def initialize_memory(
        self,
        batch_size,
        device
    ):
        return torch.zeros(
            batch_size,
            self.memory_slots,
            self.hidden_size,
            device=device
        )


    def forward(
        self,
        hidden,
        memory=None
    ):

        """
        hidden:
            [batch, seq, hidden]

        memory:
            [batch, memory_slots, hidden]

        """

        batch = hidden.size(0)


        if memory is None:
            memory = self.initialize_memory(
                batch,
                hidden.device
            )


        # Step1:
        # 当前信息编码

        encoded = self.encoder(hidden)


        # Step2:
        # 当前信息读取历史

        attended_memory, _ = self.memory_attention(
            encoded,
            memory,
            memory
        )


        fused = encoded + attended_memory


        # Step3:
        # 信息压缩

        score = self.compressor(
            fused
        )


        score = score.transpose(
            1,
            2
        )


        compressed = torch.softmax(
            score,
            dim=-1
        ) @ fused


        # compressed:
        # [batch,memory_slots,hidden]


        # Step4:
        # 螺旋更新

        old_mean = memory.mean(
            dim=1
        )

        new_mean = compressed.mean(
            dim=1
        )


        gate = self.update_gate(
            torch.cat(
                [
                    old_mean,
                    new_mean
                ],
                dim=-1
            )
        )


        gate = gate.unsqueeze(1)


        new_memory = (
            gate * compressed
            +
            (1-gate) * memory
        )


        return new_memory, fused
```

# 2. spiral_attention.py

```Python
import torch
import torch.nn as nn


class SpiralAttention(nn.Module):

    """
    当前信息 + 历史信息 Attention

    """

    def __init__(
        self,
        hidden_size,
        heads=8
    ):
        super().__init__()


        self.attention = nn.MultiheadAttention(
            hidden_size,
            heads,
            batch_first=True
        )


    def forward(
        self,
        x,
        memory
    ):


        context = torch.cat(
            [
                x,
                memory
            ],
            dim=1
        )


        out, _ = self.attention(
            x,
            context,
            context
        )


        return out
```

# 3. spiral_block.py

```Python
import torch.nn as nn

from spiral_memory import SpiralMemory
from spiral_attention import SpiralAttention



class SpiralBlock(nn.Module):


    def __init__(
        self,
        hidden_size
    ):
        super().__init__()


        self.attention = SpiralAttention(
            hidden_size
        )


        self.memory = SpiralMemory(
            hidden_size
        )


        self.ffn = nn.Sequential(

            nn.Linear(
                hidden_size,
                hidden_size*4
            ),

            nn.GELU(),

            nn.Linear(
                hidden_size*4,
                hidden_size
            )

        )


        self.norm1 = nn.LayerNorm(
            hidden_size
        )

        self.norm2 = nn.LayerNorm(
            hidden_size
        )



    def forward(
        self,
        x,
        memory=None
    ):


        attn = self.attention(
            x,
            memory if memory is not None else x
        )


        x = self.norm1(
            x + attn
        )


        new_memory, memory_feature = self.memory(
            x,
            memory
        )


        x = self.norm2(
            x + self.ffn(memory_feature)
        )


        return x, new_memory
```

# 4. model.py

```Python
import torch.nn as nn

from spiral_block import SpiralBlock



class InformationSpiralTransformer(nn.Module):


    def __init__(
        self,
        vocab_size,
        hidden_size=512,
        layers=6
    ):

        super().__init__()


        self.embedding = nn.Embedding(
            vocab_size,
            hidden_size
        )


        self.blocks = nn.ModuleList(
            [
                SpiralBlock(hidden_size)
                for _ in range(layers)
            ]
        )


        self.output = nn.Linear(
            hidden_size,
            vocab_size
        )



    def forward(
        self,
        tokens
    ):


        x = self.embedding(tokens)


        memory=None


        for block in self.blocks:

            x,memory = block(
                x,
                memory
            )


        logits=self.output(
            x
        )


        return logits
```
